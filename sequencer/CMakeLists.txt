cmake_minimum_required(VERSION 3.13)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH to your Raspberry Pi Pico SDK directory")
endif()
set(PICO_BOARD pico CACHE STRING "RP2040 with external flash")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(t12_sequencer VERSION 0.1.1 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
# Reserve 18 sectors for eight mirrored sequence slots and mirrored settings.
set(PICO_FLASH_SIZE_BYTES "2023424")
pico_sdk_init()
add_executable(t12_sequencer src/main.cpp src/model.cpp src/engine.cpp src/hardware.cpp src/display.cpp src/graphics.cpp src/ui.cpp src/storage.cpp)
target_include_directories(t12_sequencer PRIVATE src)
target_compile_definitions(t12_sequencer PRIVATE PICO_STACK_SIZE=0x1000 PICO_CORE1_STACK_SIZE=0x1000)
set_source_files_properties(src/main.cpp src/model.cpp src/engine.cpp src/hardware.cpp src/display.cpp src/graphics.cpp src/ui.cpp src/storage.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic")
target_link_libraries(t12_sequencer pico_stdlib pico_multicore pico_flash hardware_i2c hardware_spi hardware_adc hardware_pwm hardware_pio hardware_clocks hardware_flash)
pico_generate_pio_header(t12_sequencer ${CMAKE_CURRENT_LIST_DIR}/src/rgb.pio)
pico_enable_stdio_usb(t12_sequencer 1)
pico_enable_stdio_uart(t12_sequencer 0)
pico_set_program_name(t12_sequencer "T12 Sequencer MVP")
pico_set_program_version(t12_sequencer "0.1.1")
pico_add_extra_outputs(t12_sequencer)
